\noindent\textbf{Operators and conventions}
\medskip

\begin{longtable}{@{}L{0.23\textwidth}L{0.68\textwidth}@{}}
\endfirsthead
\multicolumn{2}{@{}l}{\textbf{Operators and conventions (continued)}}\\[0.5ex]
\endhead
$\Expect[\,\cdot\,]$ & ensemble-expectation operator; $\Expect^{\alpha}[\,\cdot\,]$ denotes the cycle-frequency-$\alpha$ Fourier coefficient of a time-varying ensemble mean \\
$\widehat{(\,\cdot\,)}$ & finite-record estimate or finite-record Fourier evaluation, as defined locally \\
$(\,\cdot\,)^*$ & complex conjugation \\
$\ast$ & convolution \\
$\Pr(\,\cdot\,)$ & probability operator \\
$\Real(\,\cdot\,),\,\Imag(\,\cdot\,)$ & real- and imaginary-part operators \\
$\mathrm{e}$ & base of the natural logarithm \\
$\mathrm{j}$ & imaginary unit, $\mathrm{j}^2=-1$ \\
$\mathrm{d}t,\,\mathrm{d}f$ & differentials with upright $\mathrm{d}$ \\
\texttt{aclk}, \texttt{aresetn} & AXI4-Stream clock and active-low reset \\
\texttt{s\_axis}, \texttt{m\_axis} & AXI4-Stream input and output channels, respectively \\
\texttt{tdata}, \texttt{tvalid}, \texttt{tready}, \texttt{tlast} & AXI4-Stream payload, validity, readiness, and record-boundary signals \\
$X,\,Y$ versus $x,\,y$ & uppercase denotes random processes; lowercase denotes realizations, finite records, and datapath sequences \\
$n,\,v,\,k,\,\ell$ & zero-based mathematical indices unless a one-based MATLAB index is stated explicitly \\
\end{longtable}
